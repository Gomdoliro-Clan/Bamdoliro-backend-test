// ---- FormRepositoryImpl 발췌 ----
    @Override
    public List<Form> findApprovedSpecialForm() {
        return queryFactory
                .selectFrom(form)
                .where(
                        form.status.eq(FormStatus.APPROVED)
                                .and(isSpecialType())
                )
                .orderBy(form.score.firstRoundScore.desc())
                .fetch();
    }

    @Override
    public List<Form> findApprovedRegularForm() {
        return queryFactory
                .selectFrom(form)
                .where(
                        form.status.eq(FormStatus.APPROVED)
                                .and(isRegularType())
                )
                .orderBy(form.score.firstRoundScore.desc())
                .fetch();
    }

    @Override
    public List<Form> findApprovedSupernumeraryForm() {
        return queryFactory
                .selectFrom(form)
                .where(
                        form.status.eq(FormStatus.APPROVED)
                                .and(isSupernumeraryType())
                )
                .orderBy(form.score.firstRoundScore.desc())
                .fetch();
    }

// ---- Form 엔티티 발췌: 점수와 기본키 ----
    @Column(name = "form_id")
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Id
    private Long id;

    @Embedded
    private Score score;   // firstRoundScore: Double
